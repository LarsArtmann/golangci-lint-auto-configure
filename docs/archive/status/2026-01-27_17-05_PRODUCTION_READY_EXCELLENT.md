#!/bin/bash
# scripts/verify_status_report.sh

verify_file_exists() {
    local file="$1"
    local description="$2"

    if [ ! -f "$file" ]; then
        echo "❌ ERROR: $description does not exist: $file"
        return 1
    fi

    if [ ! -s "$file" ]; then
        echo "❌ ERROR: $description is empty: $file"
        return 1
    fi

    echo "✅ Verified: $description ($file, $(wc -l < "$file") lines)"
    return 0
}
